    <script>
      // 1. 頁面載入時直接啟動
      window.addEventListener("load", () => {
        // 直接執行圖表初始化
        initCharts();
      });
      // 2. 僅保留切換頁籤功能
      function switchTab(type) {
        const isToday = type === "today";
        document.getElementById("content-today").style.display = isToday
          ? "block"
          : "none";
        document.getElementById("content-history").style.display = isToday
          ? "none"
          : "block";
        document
          .getElementById("tab-today")
          .classList.toggle("active-tab", isToday);
        document
          .getElementById("tab-history")
          .classList.toggle("active-tab", !isToday);

        // 切換到歷史時，如果圖表未跑過寬度計算，可在此重新觸發 initCharts 或調整
      }

      // --- 照顧日誌邏輯 ---
      function switchTab(type) {
        const isToday = type === "today";
        document.getElementById("content-today").style.display = isToday
          ? "block"
          : "none";
        document.getElementById("content-history").style.display = isToday
          ? "none"
          : "block";

        document
          .getElementById("tab-today")
          .classList.toggle("active-tab", isToday);
        document
          .getElementById("tab-history")
          .classList.toggle("active-tab", !isToday);
      }

      // --- 圖表初始化 (使用 Chart.js 原生藍色) ---
      Chart.defaults.color = "#777";
      Chart.defaults.font.family =
        '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto';

      let pieChart, lineChart;

      function initCharts() {
        // 防止重複初始化
        if (pieChart) pieChart.destroy();
        if (lineChart) lineChart.destroy();

        // (1) 今日情緒圓餅圖
        const ctxPie = document.getElementById("pieChart").getContext("2d");
        pieChart = new Chart(ctxPie, {
          type: "pie",
          data: {
            labels: ["開心", "平靜", "焦慮"],
            datasets: [
              {
                data: [60, 30, 10],
                // 使用更柔和的調色盤
                backgroundColor: ["#4caf50", "#3b77c4", "#ffce56"],
                borderWidth: 0,
              },
            ],
          },
          options: {
            maintainAspectRatio: false,
            plugins: {
              legend: {
                position: "bottom",
                labels: { boxWidth: 12, padding: 20 },
              },
              title: {
                display: true,
                text: "今日情緒分佈統計",
                font: { size: 16 },
              },
            },
          },
        });

        // (2) 歷史趨勢折線圖 - 迴圈生成 mock data 與背景
        const daysCount = 14;
        const labels = [];
        const posData = [];
        const negData = [];
        const dayAnnotations = {};

        for (let i = 0; i < daysCount; i++) {
          labels.push(`4/${i + 1}`);
          posData.push(Math.floor(Math.random() * 6) + 7); // 7-12 之間
          negData.push(Math.floor(Math.random() * 4) + 2); // 2-5 之間

          // 每隔一天產生灰色背景色塊 (0, 2, 4...)
          if (i % 2 === 0) {
            dayAnnotations[`box${i}`] = {
              type: "box",
              xMin: i - 0.5,
              xMax: i + 0.5,
              backgroundColor: "rgba(0, 0, 0, 0.03)",
              borderWidth: 0,
              z: -1,
            };
          }
        }

        // 動態計算 Canvas 寬度 (每6天一個螢幕寬度)
        const chartCanvas = document.getElementById("lineChart");
        const chartWrapper = chartCanvas.parentElement;
        const visibleDays = 6;
        if (labels.length > visibleDays) {
          const newWidth = (labels.length / visibleDays) * 100;
          chartWrapper.style.width = newWidth + "%";
        } else {
          chartWrapper.style.width = "100%";
        }

        const ctxLine = chartCanvas.getContext("2d");
        lineChart = new Chart(ctxLine, {
          type: "line",
          data: {
            labels: labels,
            datasets: [
              {
                label: "正向",
                data: posData,
                borderColor: "#4CAF50",
                tension: 0.3,
                pointBackgroundColor: "#fff",
              },
              {
                label: "負向",
                data: negData,
                borderColor: "#F44336",
                tension: 0.3,
                pointBackgroundColor: "#fff",
              },
            ],
          },
          options: {
            maintainAspectRatio: false,
            responsive: true,
            // 大範圍點擊
            interaction: { mode: "index", intersect: false },
            plugins: {
              annotation: { annotations: dayAnnotations },
              legend: { display: true, position: "bottom" },
              title: {
                display: true,
                text: "最近 14 天情緒趨勢圖",
                font: { size: 16 },
              },
            },
            scales: {
              x: { grid: { display: false } },
              y: {
                beginAtZero: true,
                suggestedMax: 15,
                title: { display: true, text: "發生次數" },
              },
            },
            onClick: (evt) => {
              const points = lineChart.getElementsAtEventForMode(
                evt,
                "index",
                { intersect: false },
                true,
              );
              if (points.length > 0) {
                const index = points[0].index;
                const date = lineChart.data.labels[index];
                document.getElementById("selected-history-date").innerText =
                  `📅 ${date} 歷史活動明細`;

                // 模擬更新表格內容
                const tbody = document.querySelector(
                  "#history-detail-table tbody",
                );
                tbody.innerHTML = `<tr><td>14:00</td><td>測試數據活動 ${date}</td><td>😊 正向</td></tr>`;
              }
            },
          },
        });
      }
    </script>
